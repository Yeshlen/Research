% Pipeline overview. Needs: \usepackage{tikz} \usetikzlibrary{positioning,arrows.meta}
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[
  node distance=9mm and 5mm,
  box/.style={draw, rounded corners=2pt, align=center, minimum height=15mm, text width=23mm,
              inner sep=2pt, font=\footnotesize, fill=white},
  data/.style={box, rounded corners=0pt, fill=black!7},
  sub/.style={font=\scriptsize},
  arrow/.style={-{Stealth[length=2mm]}, thick},
]
% Row 1: generation and feature extraction
\node[data] (cfg) {Configuration\\[-1pt] {\scriptsize parameters\\[-2pt] and seed}};
\node[box, right=of cfg] (sim) {Agent-based\\[-1pt] simulator\\[-1pt] {\scriptsize customers and\\[-2pt] episodes}};
\node[data, right=of sim] (log) {Event log\\[-1pt] {\scriptsize timestamped\\[-2pt] app events}};
\node[box, right=of log] (feat) {Feature\\[-1pt] extraction\\[-1pt] {\scriptsize streaming,\\[-2pt] past events only}};
\node[data, right=of feat] (tab) {Feature table\\[-1pt] {\scriptsize stage A and\\[-2pt] stage B}};
% Row 2: labels, scoring, evaluation
\node[data, below=of sim] (truth) {Ground truth\\[-1pt] {\scriptsize scam label and\\[-2pt] scenario type}};
\node[box, below=of tab] (score) {Risk scoring\\[-1pt] {\scriptsize rules, anomaly\\[-2pt] models, hybrid}};
\node[box, below=of feat] (eval) {Evaluation\\[-1pt] {\scriptsize time-based\\[-2pt] split}};
\node[box, below=of score] (dash) {Dashboard\\[-1pt] {\scriptsize alerts, bands\\[-2pt] and reasons}};

\draw[arrow] (cfg) -- (sim);
\draw[arrow] (sim) -- (log);
\draw[arrow] (log) -- (feat);
\draw[arrow] (feat) -- (tab);
\draw[arrow] (sim) -- (truth);
\draw[arrow] (tab) -- (score);
\draw[arrow] (score) -- (eval);
\draw[arrow] (score) -- (dash);
\draw[arrow, dashed] (truth) -- node[above, sub] {labels} node[below, sub] {evaluation only} (eval);
\end{tikzpicture}
\caption{Overview of the prototype pipeline. The simulator wrote a raw event log and a separate ground-truth file. Features were computed by replaying the event log in time order, so each voucher was described using only earlier events. Labels were used for evaluation only and were never given to the scoring models as inputs.}
\label{fig:pipeline}
\end{figure}
